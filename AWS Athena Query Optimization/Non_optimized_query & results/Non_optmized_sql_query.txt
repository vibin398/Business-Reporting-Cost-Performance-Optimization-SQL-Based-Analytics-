SELECTyear(COALESCE(try(date_parse(order_date, '%Y-%m-%dT%H:%i:%s.%f')),try(date_parse(order_date, '%Y-%m-%dT%H:%i:%s')))) AS order_year,month(COALESCE(try(date_parse(order_date, '%Y-%m-%dT%H:%i:%s.%f')),try(date_parse(order_date, '%Y-%m-%dT%H:%i:%s')))) AS order_month,COUNT(*) AS order_count,SUM(total_amount) AS total_salesFROM investigation.orders_rawWHERE status = 'Completed'GROUP BYyear(COALESCE(try(date_parse(order_date, '%Y-%m-%dT%H:%i:%s.%f')),try(date_parse(order_date, '%Y-%m-%dT%H:%i:%s')))),month(COALESCE(try(date_parse(order_date, '%Y-%m-%dT%H:%i:%s.%f')),try(date_parse(order_date, '%Y-%m-%dT%H:%i:%s'))))ORDER BY order_year, order_month;
